cmake_minimum_required(VERSION 3.22)
project(SDMood VERSION 0.1.0)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(SDMood COMPANY_NAME "Open Circuit" BUNDLE_ID com.opencircuit.mood PLUGIN_MANUFACTURER_CODE OpCi PLUGIN_CODE Mood FORMATS VST3 Standalone PRODUCT_NAME "SD MOOD" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE)
target_sources(SDMood PRIVATE Source/PluginProcessor.cpp)
target_compile_features(SDMood PRIVATE cxx_std_17)
target_compile_definitions(SDMood PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(SDMood PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
